
### Overview
A `SourceConnector` interface shape that provide a common ground.
	1. Bespoke extraction of data
	2. Consistent chunking of artifacts into regions
		1. The split functionality must ensure pre-processing of artifacts to ignore all stylistic changes, and only detect diffs in region hashes when content based changes are made to the source material
	3. Generation of relational and provenance graph

```
interface SourceConnector {
      // --- Metadata declaration (read by the engine, not branched on inside) ---
      val laneId: String                          // "github" | "slack" | "notion" | ...
      val sourceClass: SourceClass                // live_pointer | hybrid | native
      val defaultConfidence: Confidence

      // --- Acquisition (already in your draft) ---
      fun acquire(
          workspace: WorkspaceId,
          cursor: SyncCursor?
      ): Flow<RawArtifact>

      // --- Pass 0 contributions (new scope) ---
      fun split(artifact: RawArtifact): List<Region>
      fun extractProvenance(
          artifact: RawArtifact,
          regions: List<Region>
      ): List<ProvenanceEdge>
  }
```

```
 data class RawArtifact(
      val laneId: String,
      val sourceClass: SourceClass,
      val canonicalUri: String,
      val displayName: String,
      val upstreamRef: UpstreamRef?,          // null for native/digest
      val payload: ByteArray,                  // or streaming handle
      val contentType: ContentType,            // CODE_KOTLIN | MARKDOWN | NOTION_BLOCK | SLACK_THREAD | ...
      val fetchedAt: Instant
  )

  data class Region(
      val regionId: String,                    // stable slug
      val locator: Locator,                    // sealed: LineRange | Symbol | Heading | Ts | BlockId | XPath
      val regionClass: RegionClass,            // UPSTREAM | NATIVE
      val text: String                         // normalized for hashing
  )

  data class ProvenanceEdge(
      val fromSource: SourceId, val fromRegion: RegionId?,
      val toSource: SourceId,   val toRegion: RegionId?,
      val edgeKind: EdgeKind,                  // CALLS | IMPORTS | CITES | MENTIONS | QUOTES | SUPERSEDES | BLOCKS
      val extractedBy: String                  // for debugging — "tree-sitter@0.22" | "markdown-parser@2.4"
  )
```
